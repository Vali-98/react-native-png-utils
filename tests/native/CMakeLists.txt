cmake_minimum_required(VERSION 3.16)
project(PngUtilsNativeTests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(PNGUTIL_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
set(PNGUTIL_SANITIZERS "address,undefined" CACHE STRING "Clang sanitizers to enable")
option(PNGUTIL_NEON "Enable ARM64 NEON" ON)
option(PNGUTIL_BASE64_CHECK "Enable complete validation and safe unaligned loads" ON)
set(PNGUTIL_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../cpp/HybridPngUtils.cpp" CACHE FILEPATH "Implementation to test/benchmark")
set(PNGUTIL_BASE64_DIR "${CMAKE_CURRENT_LIST_DIR}/../../cpp/turbob64" CACHE PATH "Base64 source directory")
if(PNGUTIL_SANITIZE)
    add_compile_options(-fsanitize=${PNGUTIL_SANITIZERS} -fno-omit-frame-pointer -fno-sanitize-recover=all)
    add_link_options(-fsanitize=${PNGUTIL_SANITIZERS})
endif()
find_package(ZLIB REQUIRED)
add_library(pngutils-core STATIC ${PNGUTIL_SOURCE}
    ${PNGUTIL_BASE64_DIR}/turbob64c.c ${PNGUTIL_BASE64_DIR}/turbob64d.c)
target_include_directories(pngutils-core PUBLIC support ../../cpp)
# memcpy loads avoid undefined unaligned accesses in the vendored scalar code.
if(PNGUTIL_BASE64_CHECK)
    target_compile_definitions(pngutils-core PRIVATE B64CHECK UA_MEMCPY)
endif()
if(PNGUTIL_NEON AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)$")
    target_sources(pngutils-core PRIVATE ${PNGUTIL_BASE64_DIR}/turbob64v128.c)
    target_compile_definitions(pngutils-core PRIVATE PNGUTIL_ENABLE_NEON)
endif()
target_link_libraries(pngutils-core PUBLIC ZLIB::ZLIB)
add_executable(pngutils-tests tests.cpp check-pixel.cpp)
target_link_libraries(pngutils-tests PRIVATE pngutils-core)
add_executable(pngutils-benchmark benchmark.cpp)
target_link_libraries(pngutils-benchmark PRIVATE pngutils-core)
enable_testing()
add_test(NAME pngutils-tests COMMAND pngutils-tests)

cmake_minimum_required(VERSION 3.16)
project(AudioConversionTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(audio-convert convert.cpp ../../cpp/audio/WavWriter.cpp)
target_include_directories(audio-convert PRIVATE ../../cpp/audio)
if(ANDROID)
  target_sources(audio-convert PRIVATE ../../cpp/audio/AudioDecoder.android.cpp)
  target_link_libraries(audio-convert PRIVATE mediandk)
elseif(APPLE)
  target_sources(audio-convert PRIVATE ../../cpp/audio/AudioDecoder.ios.cpp)
  target_link_libraries(audio-convert PRIVATE "-framework AudioToolbox" "-framework CoreFoundation")
else()
  message(FATAL_ERROR "Audio conversion uses Android or Apple platform codecs")
endif()
if(NOT ANDROID)
  add_executable(audio-writer-tests writer-tests.cpp ../../cpp/audio/WavWriter.cpp)
  target_include_directories(audio-writer-tests PRIVATE ../../cpp/audio)
  enable_testing()
  add_test(NAME audio-writer-tests COMMAND audio-writer-tests)
endif()

if(APPLE)
  add_executable(audio-android-loop-tests android-loop-tests.cpp
    ../../cpp/audio/AudioDecoder.android.cpp ../../cpp/audio/WavWriter.cpp)
  target_include_directories(audio-android-loop-tests PRIVATE fake-ndk ../../cpp/audio)
  target_compile_definitions(audio-android-loop-tests PRIVATE PNGUTIL_TEST_ANDROID_DECODER)
  add_test(NAME audio-android-loop-tests COMMAND audio-android-loop-tests)
endif()
